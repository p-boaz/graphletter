**Goal:** Classify every file under `data/` as `unmodified-from-upstream`, `flagged-for-review`, or `graphletter-authored` using **git history alone** (no upstream fetch — SCF doesn't publish per-tab CSVs on GitHub and the `.xlsx` is behind a click-through). Emit `data/LICENSE_AUDIT.json` (machine-readable) and `data/README.md` (attribution + classification). The seed plan's Task 7 reads the JSON to gate the AO-dump task.

**Architecture:** Two-pass. (1) For each `data/*` file, walk `git log --follow --name-status` and tag each commit as `A` (add — initial commit) or `M` (modify — content change). Add-only history → `unmodified`; any modify commit → `flagged-for-review` with the offending commit SHA + diff summary surfaced to peter. (2) Authoring-source map (small lookup table) records which files are upstream-SCF redistributions vs. graphletter-authored fixtures; the script combines history + lookup into a final classification.

**Tech Stack:** Node 20+, `tsx`, `node:child_process` (`git log`), `node:fs`, no network access required, no upstream fetch.

import { execFileSync } from "node:child_process";
import { readdir, stat, writeFile } from "node:fs/promises";
import { join, resolve } from "node:path";
import { fileURLToPath } from "node:url";

export interface FileOp {
  status: "A" | "M" | "D" | "R" | "C";
  path: string;
  oldPath?: string;
}

export interface GitCommit {
  sha: string;
  author: string;
  date: string;
  subject: string;
  fileOps: FileOp[];
}

/**
 * Parse the output of:
 *   git log --follow --name-status --format=%H%x09%ae%x09%aI%x09%s -- <path>
 *
 * Output format: one commit per record, separated by blank lines.
 * Each record:
 *   line 1: SHA\tauthor_email\tISO_date\tsubject
 *   line 2..n: status\tpath  (or for renames: R100\told\tnew)
 */
export function parseGitLogNameStatus(raw: string): GitCommit[] {
  const records = raw
    .replace(/\r/g, "")
    .split(/\n\s*\n/)
    .filter((r) => r.trim().length > 0);
  return records.map((rec) => {
    const lines = rec.split("\n");
    const [sha, author, date, ...subjParts] = lines[0].split("\t");
    const subject = subjParts.join("\t");
    const fileOps: FileOp[] = [];
    for (const line of lines.slice(1)) {
      if (!line) continue;
      const parts = line.split("\t");
      const rawStatus = parts[0];
      const firstChar = rawStatus.charAt(0);
      if (firstChar === "R" || firstChar === "C") {
        fileOps.push({
          status: firstChar as "R" | "C",
          oldPath: parts[1],
          path: parts[2],
        });
      } else if (["A", "M", "D"].includes(firstChar)) {
        fileOps.push({
          status: firstChar as "A" | "M" | "D",
          path: parts[1],
        });
      }
    }
    return { sha, author, date, subject, fileOps };
  });
}

export type Classification =
  | "unmodified"
  | "flagged-for-review"
  | "graphletter-authored"
  | "documentation";

export interface ClassifyResult {
  classification: Classification;
  reason: string;
  flaggedCommits: string[];
}

/**
 * Classify a file based on its commit history alone.
 *
 * Rules:
 *  - 0 commits: classification is "graphletter-authored" (untracked or new in this branch).
 *  - All commits status === "A" (add) or "R" (rename without content change): "unmodified".
 *  - Any commit with status === "M": "flagged-for-review" with SHA list.
 *  - "D" commits at the end mean the file was deleted — ignore for this analysis.
 */
export function classifyByHistory(commits: GitCommit[], file: string): ClassifyResult {
  const opsForFile = commits.flatMap((c) =>
    c.fileOps
      .filter((op) => op.path === file || op.oldPath === file)
      .map((op) => ({ ...op, sha: c.sha }))
  );
  if (opsForFile.length === 0) {
    return {
      classification: "graphletter-authored",
      reason: "No git history found for this file (untracked or new).",
      flaggedCommits: [],
    };
  }
  const modifyCommits = opsForFile.filter((op) => op.status === "M");
  if (modifyCommits.length === 0) {
    return {
      classification: "unmodified",
      reason: `Git history shows only ${opsForFile.length} non-modify op(s) (A/R/C).`,
      flaggedCommits: [],
    };
  }
  return {
    classification: "flagged-for-review",
    reason: `${modifyCommits.length} commit(s) modified file contents; manual review required to confirm whether the modification is a derivative work under CC BY-ND.`,
    flaggedCommits: modifyCommits.map((c) => c.sha),
  };
}

/**
 * Authoring source per file. This is the lookup half of the audit.
 *
 *  - "upstream-scf": redistributed verbatim from securecontrolsframework.com
 *    (version 2025.1.1). Subject to CC BY-ND 4.0; requires attribution.
 *  - "graphletter":  authored in this repo. Subject to the repo's own license.
 *  - "documentation": markdown/PDF docs; not subject to the SCF license terms.
 */
export type AuthoringSource = "upstream-scf" | "graphletter" | "documentation";

export const AUTHORING: Record<string, AuthoringSource> = {
  "controls.csv": "upstream-scf",
  "full_scf.csv": "upstream-scf",
  "full_scf_rev.csv": "upstream-scf",
  "Authoritative Sources.csv": "upstream-scf",
  "Domains and Principles.csv": "upstream-scf",
  "risks.csv": "upstream-scf",
  "threats.csv": "upstream-scf",
  "ERL.csv": "upstream-scf",
  "Assessment_objectives.csv": "upstream-scf",
  "scf_evidence_request_list_rows.csv": "graphletter",
  "anthropic-controls.pdf": "graphletter",
  "scf_data_import_strategy.md": "documentation",
  "scf_import_usage_guide.md": "documentation",
};

export interface AuditEntry {
  file: string;
  authoringSource: AuthoringSource;
  classification: Classification;
  reason: string;
  flaggedCommits: string[];
}

export interface AuditReport {
  generatedAt: string;
  scfVersion: "2025.1.1";
  verdict: "clean" | "attribution-required" | "derivative-blocker";
  entries: AuditEntry[];
}

export function gitLogForFile(repoRoot: string, file: string): GitCommit[] {
  const out = execFileSync(
    "git",
    ["log", "--follow", "--name-status", "--format=%H%x09%ae%x09%aI%x09%s", "--", file],
    { cwd: repoRoot, encoding: "utf8" }
  );
  return parseGitLogNameStatus(out);
}

export async function runAudit(repoRoot: string): Promise<AuditReport> {
  const dataDir = join(repoRoot, "data");
  const fileNames = (await readdir(dataDir)).sort();
  const entries: AuditEntry[] = [];

  for (const name of fileNames) {
    const fullPath = join(dataDir, name);
    const st = await stat(fullPath);
    if (st.isDirectory()) continue;
    const authoringSource = AUTHORING[name] ?? "graphletter";
    if (authoringSource === "documentation") {
      entries.push({
        file: name,
        authoringSource,
        classification: "documentation",
        reason: "Markdown/PDF documentation; not subject to SCF redistribution rules.",
        flaggedCommits: [],
      });
      continue;
    }
    if (authoringSource === "graphletter") {
      entries.push({
        file: name,
        authoringSource,
        classification: "graphletter-authored",
        reason: "Authored in graphletter (not a verbatim upstream redistribution).",
        flaggedCommits: [],
      });
      continue;
    }
    const commits = gitLogForFile(repoRoot, `data/${name}`);
    const result = classifyByHistory(commits, `data/${name}`);
    entries.push({
      file: name,
      authoringSource,
      classification: result.classification,
      reason: result.reason,
      flaggedCommits: result.flaggedCommits,
    });
  }

  const hasFlagged = entries.some((e) => e.classification === "flagged-for-review");
  const hasUpstreamCopy = entries.some(
    (e) => e.authoringSource === "upstream-scf" && e.classification === "unmodified"
  );
  const verdict: AuditReport["verdict"] = hasFlagged
    ? "derivative-blocker"
    : hasUpstreamCopy
      ? "attribution-required"
      : "clean";

  return {
    generatedAt: new Date().toISOString(),
    scfVersion: "2025.1.1",
    verdict,
    entries,
  };
}

export function renderReadme(report: AuditReport): string {
  const lines: string[] = [];
  lines.push("# graphletter `data/` license");
  lines.push("");
  lines.push(`Generated by \`pnpm audit:scf-license\` on ${report.generatedAt}.`);
  lines.push(`Verdict: **${report.verdict}**.`);
  lines.push("");
  lines.push("## Attribution");
  lines.push("");
  lines.push(
    "Files authored upstream-SCF and classified \\`unmodified\\` are redistributed under " +
      "[Creative Commons Attribution-NoDerivatives 4.0 International (CC BY-ND 4.0)](https://creativecommons.org/licenses/by-nd/4.0/), " +
      `sourced from the Secure Controls Framework, version ${report.scfVersion} ` +
      "(https://securecontrolsframework.com)."
  );
  lines.push("");
  lines.push("## Per-file classification");
  lines.push("");
  lines.push("| File | Authoring | Classification | Reason |");
  lines.push("|---|---|---|---|");
  for (const e of report.entries) {
    const reason = e.reason.replace(/\|/g, "\\|");
    lines.push(`| \`${e.file}\` | ${e.authoringSource} | ${e.classification} | ${reason} |`);
  }
  lines.push("");
  if (report.verdict === "derivative-blocker") {
    lines.push("## ⚠️ Action required");
    lines.push("");
    lines.push(
      "One or more files are flagged for review. CC BY-ND 4.0 prohibits " +
        "redistribution of derivative works. Inspect each flagged commit; if the " +
        "modification is non-trivial, either revert to upstream content or remove " +
        "the file from the repository before publishing a release."
    );
    lines.push("");
    lines.push("### Flagged commits");
    lines.push("");
    for (const e of report.entries) {
      if (e.flaggedCommits.length === 0) continue;
      lines.push(
        `- \`${e.file}\` — ${e.flaggedCommits.map((s) => `\`${s.slice(0, 7)}\``).join(", ")}`
      );
    }
  }
  return lines.join("\n") + "\n";
}

async function main() {
  const repoRoot = fileURLToPath(new URL("..", import.meta.url));
  const report = await runAudit(repoRoot);
  const jsonPath = join(repoRoot, "data", "LICENSE_AUDIT.json");
  await writeFile(jsonPath, JSON.stringify(report, null, 2) + "\n");
  const mdPath = join(repoRoot, "data", "README.md");
  await writeFile(mdPath, renderReadme(report));
  console.log(`Wrote ${jsonPath}`);
  console.log(`Wrote ${mdPath}`);
  console.log(`Verdict: ${report.verdict}`);
  for (const e of report.entries) {
    const tag = e.classification.padEnd(22);
    console.log(`  [${tag}] ${e.file} — ${e.reason}`);
  }
  if (report.verdict === "derivative-blocker") {
    console.log("");
    console.log("⚠ Surface the flagged commits to peter before continuing.");
    process.exitCode = 2;
  }
}

if (resolve(fileURLToPath(import.meta.url)) === resolve(process.argv[1] ?? "")) {
  main().catch((err) => {
    console.error(err);
    process.exit(1);
  });
}
