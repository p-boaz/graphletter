**Goal:** Make a fork of graphletter reproducible from a clean Supabase project. After `pnpm dlx supabase db push && pnpm seed`, all 17 SCF reference tables hold rows within ±1 % of prod row counts and the dashboard's gap-guidance feature renders against real data.

**Architecture:** Three layers. (1) Schema — already handled by `supabase db push` (commit `fbbdb8e` reconciled the ledger). (2) Reference data — a new orchestrator `scripts/seed-all.ts` runs five steps in order: (a) extracted core-SCF writer (`lib/scf/writer.ts`, used by both the orchestrator and the existing in-app uploader); (b) the existing `scripts/import-scf-data.js` as a subprocess (handles risks / threats / maturity / mappings, well-tested, do not absorb); (c) a new ERL importer reading `data/scf_evidence_request_list_rows.csv`; (d) a new Assessment Objectives importer reading `data/scf_assessment_objectives_rows.csv` (gated on the license audit verdict); (e) a derived `scf_control_evidence_mappings` populator that joins ERL × controls. (3) Graphletter-authored fixtures — `scf_control_integrations` (4 rows) ships as a seed migration, not as a CSV importer.

**Tech Stack:** Node 20+, `tsx`, `@supabase/supabase-js@^2.88`, `csv-parse@^6`, `dotenv@^17`, `node:child_process` (subprocess fan-out), `node:test` (verification harness), Supabase MCP (sandbox bootstrap), Supabase CLI (`pnpm dlx supabase db push` + `pnpm dlx supabase db reset` — pooler login bypassed by setting `SUPABASE_DB_PASSWORD`).

- [ ] **Step 1: Create the importer**

```ts
import { parse as parseCsv } from "csv-parse/sync";
import { readFileSync } from "node:fs";
import { resolve } from "node:path";
import { createClient, type SupabaseClient } from "@supabase/supabase-js";
import { config as loadDotenv } from "dotenv";

export interface ErlRow {
  id: string;
  erl_id: string;
  area_of_focus: string;
  documentation_artifact: string;
  artifact_description: string;
  scf_control_mappings: string[];
  scf_version: string;
}

const REQUIRED_HEADERS = [
  "id",
  "erl_id",
  "area_of_focus",
  "documentation_artifact",
  "artifact_description",
  "scf_control_mappings",
  "scf_version",
] as const;

export function parseErlCsv(csv: string): ErlRow[] {
  const records = parseCsv(csv, {
    columns: true,
    skip_empty_lines: true,
    bom: true,
    relax_quotes: true,
  }) as Record<string, string>[];

  if (records.length === 0) return [];
  for (const h of REQUIRED_HEADERS) {
    if (!(h in records[0])) throw new Error(`ERL CSV missing required column: ${h}`);
  }

  return records.map((r) => ({
    id: r.id,
    erl_id: r.erl_id,
    area_of_focus: r.area_of_focus,
    documentation_artifact: r.documentation_artifact,
    artifact_description: r.artifact_description,
    scf_control_mappings: JSON.parse(r.scf_control_mappings) as string[],
    scf_version: r.scf_version,
  }));
}

export async function seedErl(
  supabase: SupabaseClient,
  csvPath: string
): Promise<{ inserted: number }> {
  const csv = readFileSync(csvPath, "utf8");
  const rows = parseErlCsv(csv);

  // Idempotent: delete existing rows for this scf_version, then insert.
  const versions = Array.from(new Set(rows.map((r) => r.scf_version)));
  for (const v of versions) {
    const { error } = await supabase
      .from("scf_evidence_request_list")
      .delete()
      .eq("scf_version", v);
    if (error) throw new Error(`Failed to clear ERL for version ${v}: ${error.message}`);
  }

  const batchSize = 500;
  for (let i = 0; i < rows.length; i += batchSize) {
    const batch = rows.slice(i, i + batchSize);
    const { error } = await supabase.from("scf_evidence_request_list").insert(batch);
    if (error) throw new Error(`Failed to insert ERL batch ${i / batchSize + 1}: ${error.message}`);
  }

  return { inserted: rows.length };
}

async function main() {
  loadDotenv({ path: process.env.ENV_FILE ?? ".env.sandbox" });
  const url = process.env.NEXT_PUBLIC_SUPABASE_URL;
  const key = process.env.SUPABASE_SERVICE_ROLE_KEY;
  if (!url || !key) {
    throw new Error(
      "NEXT_PUBLIC_SUPABASE_URL and SUPABASE_SERVICE_ROLE_KEY required (.env.sandbox)"
    );
  }
  const supabase = createClient(url, key, { auth: { persistSession: false } });
  const csvPath = resolve(process.cwd(), "data", "scf_evidence_request_list_rows.csv");
  const result = await seedErl(supabase, csvPath);
  console.log(`Seeded ERL: ${result.inserted} rows`);
}

if (import.meta.url === `file://${process.argv[1]}`) {
  main().catch((err) => {
    console.error(err);
    process.exit(1);
  });
}
```

```ts
import { parse as parseCsv } from "csv-parse/sync";
import { readFileSync } from "node:fs";
import { resolve } from "node:path";
import { createClient, type SupabaseClient } from "@supabase/supabase-js";
import { config as loadDotenv } from "dotenv";

export interface ErlRowMinimal {
  id: string;
  erl_id: string;
  scf_control_mappings: string[];
}

export interface DerivedMapping {
  scf_control_id: string;
  evidence_request_id: string;
  relationship_type: "direct";
  priority: number;
  is_active: true;
}

export function deriveControlEvidenceMappings(erlRows: ErlRowMinimal[]): DerivedMapping[] {
  const seen = new Set<string>();
  const out: DerivedMapping[] = [];
  for (const erl of erlRows) {
    for (const controlId of erl.scf_control_mappings) {
      const key = `${controlId}::${erl.id}`;
      if (seen.has(key)) continue;
      seen.add(key);
      out.push({
        scf_control_id: controlId,
        evidence_request_id: erl.id,
        relationship_type: "direct",
        priority: 1,
        is_active: true,
      });
    }
  }
  return out;
}

export async function seedControlEvidenceMappings(
  supabase: SupabaseClient,
  erlCsvPath: string
): Promise<{ inserted: number }> {
  const csv = readFileSync(erlCsvPath, "utf8");
  const records = parseCsv(csv, { columns: true, skip_empty_lines: true, bom: true }) as Record<
    string,
    string
  >[];
  const erlRows = records.map((r) => ({
    id: r.id,
    erl_id: r.erl_id,
    scf_control_mappings: JSON.parse(r.scf_control_mappings) as string[],
  }));
  const derived = deriveControlEvidenceMappings(erlRows);

  const { error: delErr } = await supabase
    .from("scf_control_evidence_mappings")
    .delete()
    .eq("relationship_type", "direct");
  if (delErr) throw new Error(`Failed to clear control evidence mappings: ${delErr.message}`);

  const batchSize = 500;
  for (let i = 0; i < derived.length; i += batchSize) {
    const batch = derived.slice(i, i + batchSize);
    const { error } = await supabase.from("scf_control_evidence_mappings").insert(batch);
    if (error)
      throw new Error(`Failed to insert mappings batch ${i / batchSize + 1}: ${error.message}`);
  }
  return { inserted: derived.length };
}

async function main() {
  loadDotenv({ path: process.env.ENV_FILE ?? ".env.sandbox" });
  const url = process.env.NEXT_PUBLIC_SUPABASE_URL;
  const key = process.env.SUPABASE_SERVICE_ROLE_KEY;
  if (!url || !key)
    throw new Error("NEXT_PUBLIC_SUPABASE_URL and SUPABASE_SERVICE_ROLE_KEY required");
  const supabase = createClient(url, key, { auth: { persistSession: false } });
  const csvPath = resolve(process.cwd(), "data", "scf_evidence_request_list_rows.csv");
  const result = await seedControlEvidenceMappings(supabase, csvPath);
  console.log(`Seeded control_evidence_mappings: ${result.inserted} rows`);
}

if (import.meta.url === `file://${process.argv[1]}`)
  main().catch((e) => {
    console.error(e);
    process.exit(1);
  });
```

```ts
import { readFileSync, existsSync } from "node:fs";
import { writeParsedSCF } from "../lib/scf/writer";
import { SCFParser } from "../lib/scf-parser";
import { seedErl } from "./seed-erl";
import { seedControlEvidenceMappings } from "./seed-control-evidence-mappings";

loadDotenv({ path: process.env.ENV_FILE ?? ".env.sandbox" });

const url = process.env.NEXT_PUBLIC_SUPABASE_URL;
const key = process.env.SUPABASE_SERVICE_ROLE_KEY;
if (!url || !key) {
  console.error(
    "NEXT_PUBLIC_SUPABASE_URL and SUPABASE_SERVICE_ROLE_KEY required (set ENV_FILE=.env.sandbox)"
  );
  process.exit(1);
}
const supabase = createClient(url, key, { auth: { persistSession: false } });

const dataDir = resolve(process.cwd(), "data");

const start = Date.now();

async function step(label: string, fn: () => Promise<unknown>) {
  console.log(`\n▶ ${label}`);
  const t0 = Date.now();
  const result = await fn();
  console.log(`  ✓ ${label} (${((Date.now() - t0) / 1000).toFixed(1)}s)`);
  return result;
}

async function main() {
  // 1. Core SCF (principles, domains, authoritative sources, controls, frameworks, mappings)
  await step("Core SCF (principles/domains/sources/controls/mappings)", async () => {
    const principlesCSV = readFileSync(resolve(dataDir, "Domains and Principles.csv"), "utf8");
    const authSourcesCSV = readFileSync(resolve(dataDir, "Authoritative Sources.csv"), "utf8");
    const controlsCSV = readFileSync(resolve(dataDir, "controls.csv"), "utf8");
    const parseResult = SCFParser.parseAllSCFData(principlesCSV, authSourcesCSV, controlsCSV);
    if (!parseResult.success) {
      throw new Error(`SCF parse failed: ${parseResult.errors.join("; ")}`);
    }
    // Create an import record so the writer's importId FK has something to point at.
    const { data: importRec, error: importErr } = await supabase
      .from("scf_imports")
      .insert({
        filename: "pnpm-seed",
        scf_version: parseResult.summary.version,
        import_status: "processing",
      })
      .select()
      .single();
    if (importErr) throw new Error(`Failed to create import record: ${importErr.message}`);
    const summary = await writeParsedSCF(supabase, parseResult, controlsCSV, importRec.id);
    await supabase
      .from("scf_imports")
      .update({ import_status: "completed", completed_at: new Date().toISOString() })
      .eq("id", importRec.id);
    Object.assign(stats, summary);
  });

  // 2. Risks / threats / maturity / risk-mappings / threat-mappings — keep using the existing script.
  await step("Risks / threats / maturity (subprocess: scripts/import-scf-data.js)", async () => {
    const r = spawnSync("node", ["scripts/import-scf-data.js"], {
      stdio: "inherit",
      env: { ...process.env, NEXT_PUBLIC_SUPABASE_URL: url, SUPABASE_SERVICE_ROLE_KEY: key },
    });
    if (r.status !== 0) throw new Error(`import-scf-data.js exited ${r.status}`);
  });

  // 3. ERL
  await step("Evidence Request List", async () => {
    const r = await seedErl(supabase, resolve(dataDir, "scf_evidence_request_list_rows.csv"));
    stats.evidenceRequestList = r.inserted;
  });

  // 4. Assessment Objectives (gated)
  const aoPath = resolve(dataDir, "scf_assessment_objectives_rows.csv");
  if (existsSync(aoPath)) {
    await step("Assessment Objectives", async () => {
      const { seedAssessmentObjectives } = await import("./seed-assessment-objectives");
      const r = await seedAssessmentObjectives(supabase, aoPath);
      stats.assessmentObjectives = r.inserted;
    });
  } else {
    console.warn("\n⚠  data/scf_assessment_objectives_rows.csv not found — skipping AO seed.");
    console.warn("   See SEEDING.md § Assessment Objectives for how to obtain it.");
    stats.assessmentObjectives = "skipped";
  }

  // 5. Derived control-evidence mappings
  await step("Control × Evidence mappings (derived from ERL)", async () => {
    const r = await seedControlEvidenceMappings(
      supabase,
      resolve(dataDir, "scf_evidence_request_list_rows.csv")
    );
    stats.controlEvidenceMappings = r.inserted;
  });

  console.log(`\n✓ Seed complete in ${((Date.now() - start) / 1000).toFixed(1)}s`);
  console.table(stats);
}

main().catch((e) => {
  console.error("\n✗ Seed failed:", e);
  process.exit(1);
});
```

```ts
import { createClient } from "@supabase/supabase-js";
import { readFileSync } from "node:fs";
import { resolve } from "node:path";
import { config as loadDotenv } from "dotenv";

loadDotenv({ path: process.env.ENV_FILE ?? ".env.sandbox" });

const expected = JSON.parse(
  readFileSync(resolve(process.cwd(), "data", "seed", "expected_row_counts.json"), "utf8")
) as { tables: Record<string, number> };

const url = process.env.NEXT_PUBLIC_SUPABASE_URL!;
const key = process.env.SUPABASE_SERVICE_ROLE_KEY!;
const supabase = createClient(url, key, { auth: { persistSession: false } });

const TOLERANCE = 0.01; // ±1 %

async function main() {
  let failed = 0;
  for (const [table, expectedCount] of Object.entries(expected.tables)) {
    const { count, error } = await supabase.from(table).select("*", { count: "exact", head: true });
    if (error) {
      console.error(`✗ ${table}: query error ${error.message}`);
      failed++;
      continue;
    }
    const actual = count ?? 0;
    const lower = Math.floor(expectedCount * (1 - TOLERANCE));
    const upper = Math.ceil(expectedCount * (1 + TOLERANCE));
    if (actual < lower || actual > upper) {
      console.error(
        `✗ ${table}: actual=${actual} expected=${expectedCount} (±${TOLERANCE * 100}%)`
      );
      failed++;
    } else {
      console.log(`✓ ${table}: ${actual} (expected ${expectedCount})`);
    }
  }
  if (failed > 0) {
    console.error(`\n${failed} table(s) outside tolerance.`);
    process.exit(2);
  }
  console.log("\nAll tables within tolerance.");
}

main().catch((e) => {
  console.error(e);
  process.exit(1);
});
```
